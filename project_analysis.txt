import 'dart:math' as math;

class EcuParameter {
  final String id;
  final String name;
  final int byteOffset;
  final int width;
  final double multiplier;
  final double divisor;
  final double offset;
  final int decimals;
  final bool signedComplement;
  final int bitIndex; // Если -1, то это число. Если >= 0, то битовый флаг (0 или 1)
  final String unit;

  EcuParameter({
    required this.id,
    required this.name,
    required this.byteOffset,
    required this.width,
    this.multiplier = 1.0,
    this.divisor = 1.0,
    this.offset = 0.0,
    this.decimals = 0,
    this.signedComplement = false,
    this.bitIndex = -1,
    this.unit = '',
  });

  factory EcuParameter.fromJson(Map<String, dynamic> json) {
    return EcuParameter(
      id: json['id'] as String,
      name: json['name'] as String,
      byteOffset: json['byteOffset'] as int,
      width: json['width'] as int? ?? 1,
      multiplier: (json['multiplier'] as num?)?.toDouble() ?? 1.0,
      divisor: (json['divisor'] as num?)?.toDouble() ?? 1.0,
      offset: (json['offset'] as num?)?.toDouble() ?? 0.0,
      decimals: json['decimals'] as int? ?? 0,
      signedComplement: json['signedComplement'] as bool? ?? false,
      bitIndex: json['bitIndex'] as int? ?? -1,
      unit: json['unit'] as String? ?? '',
    );
  }

  /// Метод декодирования сырых байт в готовое значение
  double? decode(List<int> payload) {
    // Защита от выхода за пределы массива
    if (byteOffset < 0 || byteOffset + width > payload.length) return null;

    int raw = 0;
    // Собираем число из байтов (Big Endian)
    for (int i = 0; i < width; i++) {
      raw = (raw << 8) | payload[byteOffset + i];
    }

    // Если это битовый флаг
    if (bitIndex >= 0) {
      return ((raw >> bitIndex) & 1).toDouble();
    }

    // Обработка отрицательных чисел (Дополнительный код / Two's complement)
    if (signedComplement) {
      final signBit = 1 << (width * 8 - 1);
      if ((raw & signBit) != 0) {
        raw -= (1 << (width * 8));
      }
    }

    // Применяем формулу
    double v = (raw * multiplier / divisor) + offset;
    
    // Округляем до нужного количества знаков
    if (decimals > 0) {
      final factor = math.pow(10, decimals);
      v = (v * factor).roundToDouble() / factor;
    } else {
      v = v.roundToDouble();
    }
    
    // Избегаем "-0.0"
    return v == 0 ? 0.0 : v;
  }
}

import 'ecu_parameter.dart';

class EcuRequestBlock {
  final String name;       // Название пакета (например, "Основные параметры")
  final String kwpCommand; // Команда для KWP2000 (например, "2101")
  final String udsCommand; // Команда для UDS (например, "222001")
  final List<EcuParameter> parameters;

  EcuRequestBlock({
    required this.name,
    required this.kwpCommand,
    required this.udsCommand,
    required this.parameters,
  });

  factory EcuRequestBlock.fromJson(Map<String, dynamic> json) {
    return EcuRequestBlock(
      name: json['name'] as String,
      kwpCommand: json['kwpCommand'] as String,
      udsCommand: json['udsCommand'] as String,
      parameters: (json['parameters'] as List)
          .map((p) => EcuParameter.fromJson(p as Map<String, dynamic>))
          .toList(),
    );
  }
}

class EcuProfile {
  final String id;
  final String name;
  final List<EcuRequestBlock> blocks;

  EcuProfile({
    required this.id,
    required this.name,
    required this.blocks,
  });

  factory EcuProfile.fromJson(Map<String, dynamic> json) {
    return EcuProfile(
      id: json['id'] as String,
      name: json['name'] as String,
      blocks: (json['blocks'] as List)
          .map((b) => EcuRequestBlock.fromJson(b as Map<String, dynamic>))
          .toList(),
    );
  }
}

enum NuanceSeverity { info, warning, critical }

class NuanceEvent {
  final String id;
  DateTime timestamp;
  NuanceSeverity severity;
  String title;
  String description;
  Map<String, String> snapshot;
  int hzCount;

  NuanceEvent({
    required this.id,
    required this.timestamp,
    required this.severity,
    required this.title,
    required this.description,
    required this.snapshot,
    this.hzCount = 1,
  });

  String get timeStr =>
      "${timestamp.hour.toString().padLeft(2, '0')}:${timestamp.minute.toString().padLeft(2, '0')}:${timestamp.second.toString().padLeft(2, '0')}";

  Map<String, dynamic> toJson() => {
        'id': id,
        'ts': timestamp.toIso8601String(),
        'sev': severity.name,
        'title': title,
        'desc': description,
        'snap': snapshot,
        'hz': hzCount,
      };

  factory NuanceEvent.fromJson(Map<String, dynamic> j) {
    final sevName = (j['sev'] ?? 'info').toString();
    final sev = NuanceSeverity.values.firstWhere(
      (e) => e.name == sevName,
      orElse: () => NuanceSeverity.info,
    );
    final rawSnap = j['snap'] as Map<String, dynamic>? ?? {};
    return NuanceEvent(
      id: (j['id'] ?? '').toString(),
      timestamp: DateTime.tryParse((j['ts'] ?? '').toString()) ?? DateTime.now(),
      severity: sev,
      title: (j['title'] ?? '').toString(),
      description: (j['desc'] ?? '').toString(),
      snapshot: rawSnap.map((k, v) => MapEntry(k, v.toString())),
      hzCount: (j['hz'] as num?)?.toInt() ?? 1,
    );
  }
}

class TrimBin {
  final String label;
  final List<double> _samples = [];
  TrimBin(this.label);

  int get count => _samples.length;
  bool get isNotEmpty => _samples.isNotEmpty;
  List<double> get rawSamples => List.unmodifiable(_samples);

  void loadSamples(List<double> saved) {
    _samples..clear()..addAll(saved.take(90));
  }

  void addSample(double stft, double ltft) {
    if (stft.abs() > 30.0 || ltft.abs() > 30.0) return;
    _samples.add(stft + ltft);
    if (_samples.length > 90) _samples.removeAt(0);
  }

  double get median {
    if (_samples.isEmpty) return 0.0;
    final sorted = List<double>.from(_samples)..sort();
    final mid = sorted.length ~/ 2;
    return sorted.length.isOdd ? sorted[mid] : (sorted[mid - 1] + sorted[mid]) / 2.0;
  }

  void clear() => _samples.clear();
}

class _O2OscilloscopePainter extends CustomPainter {
  final List<double> s1;
  final List<double> s2;
  final Color s1Color;
  final Color s2Color;
  final Color gridColor;

  _O2OscilloscopePainter({
    required this.s1,
    required this.s2,
    required this.s1Color,
    required this.s2Color,
    required this.gridColor,
  });

  @override
  void paint(Canvas canvas, Size size) {
    const maxScaleV = 1.8;
    final height = size.height;

    void drawBand(double v1, double v2, Color color) {
      final y1 = height - (v1 / maxScaleV) * height;
      final y2 = height - (v2 / maxScaleV) * height;
      canvas.drawRect(Rect.fromLTRB(0, y2, size.width, y1), Paint()..color = color);
    }
    drawBand(0.18, 0.95, AppColors.bandUdk);
    drawBand(0.59, 0.75, AppColors.bandDdk);
    drawBand(1.20, 1.70, AppColors.bandRef);

    final gridPaint = Paint()..color = AppColors.darkScopeGrid..strokeWidth = 1;
    for (int i = 1; i < 4; i++) {
      final y = height * (i / 4);
      canvas.drawLine(Offset(0, y), Offset(size.width, y), gridPaint);
    }

    void drawTrace(List<double> trace, Color color) {
      if (trace.isEmpty) return;
      final paint = Paint()
        ..color = color
        ..strokeWidth = 2.0
        ..style = PaintingStyle.stroke
        ..strokeJoin = StrokeJoin.round;

      final path = Path();
      final stepX = size.width / 99;

      for (int i = 0; i < trace.length; i++) {
        final x = i * stepX;
        final normalizedY = (trace[i] / maxScaleV).clamp(0.0, 1.0);
        final y = height - (normalizedY * height);

        if (i == 0) {
          path.moveTo(x, y);
        } else {
          path.lineTo(x, y);
        }
      }
      canvas.drawPath(path, paint);
    }

    drawTrace(s1, s1Color);
    drawTrace(s2, s2Color);
  }

  @override
  bool shouldRepaint(covariant CustomPainter oldDelegate) => true;
}

class _ScopePainter extends CustomPainter {
  final List<double> trace1;
  final List<double>? trace2;
  final double minVal;
  final double maxVal;
  final Color color1;
  final Color color2;
  final Color gridColor;

  _ScopePainter({
    required this.trace1,
    this.trace2,
    required this.minVal,
    required this.maxVal,
    required this.color1,
    required this.color2,
    required this.gridColor,
  });

  @override
  void paint(Canvas canvas, Size size) {
    // 1. Отрисовка фоновой сетки (Grid)
    final gridPaint = Paint()
      ..color = gridColor
      ..strokeWidth = 1.0;
    
    // Горизонтальные линии
    for (int i = 1; i < 4; i++) {
      final y = size.height * (i / 4);
      canvas.drawLine(Offset(0, y), Offset(size.width, y), gridPaint);
    }
    // Вертикальные линии
    for (int i = 1; i < 6; i++) {
      final x = size.width * (i / 6);
      canvas.drawLine(Offset(x, 0), Offset(x, size.height), gridPaint);
    }

    // 2. Логика отрисовки самой кривой
    void drawTrace(List<double> trace, Color color) {
      if (trace.isEmpty) return;
      final paint = Paint()
        ..color = color
        ..strokeWidth = 2.0
        ..style = PaintingStyle.stroke
        ..strokeJoin = StrokeJoin.round;

      final path = Path();
      // Выделяем ширину шага, предполагая, что на экране хранится до 100 точек
      final stepX = size.width / 99; 
      final range = maxVal - minVal;
      if (range == 0) return;

      for (int i = 0; i < trace.length; i++) {
        final x = i * stepX;
        // Нормализация (от 0.0 до 1.0) и инверсия (ось Y во Flutter идет сверху вниз)
        double normalizedY = (trace[i] - minVal) / range;
        normalizedY = normalizedY.clamp(0.0, 1.0);
        final y = size.height - (normalizedY * size.height);

        if (i == 0) {
          path.moveTo(x, y);
        } else {
          path.lineTo(x, y);
        }
      }
      canvas.drawPath(path, paint);
    }

    // Рисуем графики
    drawTrace(trace1, color1);
    if (trace2 != null) {
      drawTrace(trace2!, color2);
    }
  }

  @override
  bool shouldRepaint(covariant _ScopePainter oldDelegate) => true; 
}

class ObdState {
  final bool isConnected;
  final bool isPolling;
  final String connectedDevice;   // ← НОВОЕ
  final Map<String, double> liveData;
  final List<double> trace1;
  final List<double> trace2;
  final List<NuanceEvent> nuances;
  final Map<String, double> trimMedians;

  ObdState({
    this.isConnected = false,
    this.isPolling = false,
    this.connectedDevice = '',
    this.liveData = const {},
    this.trace1 = const [],
    this.trace2 = const [],
    this.nuances = const [],
    this.trimMedians = const {},
  });

  ObdState copyWith({
    bool? isConnected,
    bool? isPolling,
    String? connectedDevice,
    Map<String, double>? liveData,
    List<double>? trace1,
    List<double>? trace2,
    List<NuanceEvent>? nuances,
    Map<String, double>? trimMedians,
  }) {
    return ObdState(
      isConnected: isConnected ?? this.isConnected,
      isPolling: isPolling ?? this.isPolling,
      connectedDevice: connectedDevice ?? this.connectedDevice,
      liveData: liveData ?? this.liveData,
      trace1: trace1 ?? this.trace1,
      trace2: trace2 ?? this.trace2,
      nuances: nuances ?? this.nuances,
      trimMedians: trimMedians ?? this.trimMedians,
    );
  }
}

final bluetoothServiceProvider = Provider((ref) => BluetoothService());

final obdProvider = StateNotifierProvider<ObdNotifier, ObdState>((ref) {
  return ObdNotifier(ref.read(bluetoothServiceProvider));
});

class ObdNotifier extends StateNotifier<ObdState> {
  final BluetoothService _btService;
  Timer? _pollTimer;
  bool _isUdsM86 = false;
  final TripCalculator tripCalculator = TripCalculator();
  EcuProfile? _ecuProfile;

  final Map<String, TrimBin> _bins = {
    'idle': TrimBin('ХХ'),
    'low': TrimBin('Малые'),
    'part': TrimBin('Част.'),
    'high': TrimBin('Выс.'),
  };

  ObdNotifier(this._btService) : super(ObdState()) {
    _loadEcuProfile();
    _loadPersistedState();
  }

  Future<void> _loadEcuProfile() async {
    try {
      final jsonString = await rootBundle.loadString('lib/data/json_profiles/m86_uds.json');
      final jsonMap = jsonDecode(jsonString);
      _ecuProfile = EcuProfile.fromJson(jsonMap);
    } catch (e) {
      // Ошибка загрузки
    }
  }

  Future<void> _loadPersistedState() async {
    final prefs = await SharedPreferences.getInstance();
    final logJson = prefs.getString('nuance_log');
    if (logJson != null) {
      final list = jsonDecode(logJson) as List;
      state = state.copyWith(nuances: list.map((e) => NuanceEvent.fromJson(e)).toList());
    }
  }

  Future<void> _saveState() async {
    final prefs = await SharedPreferences.getInstance();
    await prefs.setString('nuance_log', jsonEncode(state.nuances.take(100).map((e) => e.toJson()).toList()));
  }

  void _logNuance(String id, String title, String desc, NuanceSeverity sev) {
    final existingIdx = state.nuances.indexWhere((e) => e.id == id);
    final updated = List<NuanceEvent>.from(state.nuances);

    if (existingIdx != -1) {
      final ex = updated.removeAt(existingIdx);
      ex.hzCount += 1;
      ex.timestamp = DateTime.now();
      updated.insert(0, ex);
    } else {
      updated.insert(0, NuanceEvent(
        id: id, timestamp: DateTime.now(), severity: sev,
        title: title, description: desc,
        snapshot: {
          if (state.liveData['rpm'] != null) 'RPM': '${state.liveData['rpm']!.toInt()}',
          if (state.liveData['ect'] != null) 'ОЖ': '${state.liveData['ect']!.toInt()}°C',
        },
      ));
    }
    state = state.copyWith(nuances: updated);
    _saveState();
  }

  void clearJournal() {
    state = state.copyWith(nuances: []);
    _saveState();
  }

  void resetTrip() {
    tripCalculator.reset();
    state = state.copyWith();
  }

  Future<bool> connect(String macAddress) async {
    final success = await _btService.connect(macAddress);
    if (!success) return false;

    await _btService.sendCommand('ATZ', timeoutMs: 2000);
    await _btService.sendCommand('ATE0');
    await _btService.sendCommand('ATL0');
    await _btService.sendCommand('ATS0');
    await _btService.sendCommand('ATH0');
    await _btService.sendCommand('ATSP6');
    await _btService.sendCommand('ATAL');
    await _btService.sendCommand('ATSH7E0');
    await _btService.sendCommand('ATCRA7E8');

    final r81 = await _btService.sendCommand('1081', timeoutMs: 800);
    if (!r81.replaceAll(' ', '').contains('5081')) {
      final r03 = await _btService.sendCommand('1003', timeoutMs: 800);
      if (r03.replaceAll(' ', '').contains('5003')) _isUdsM86 = true;
    } else {
      _isUdsM86 = false;
    }

    state = state.copyWith(isConnected: true, connectedDevice: macAddress);
    return true;
  }

  void startPolling() {
    if (!state.isConnected || state.isPolling) return;
    state = state.copyWith(isPolling: true);

    _pollTimer = Timer.periodic(const Duration(milliseconds: 500), (timer) async {
      await _pollCycle();
    });
  }

  void stopPolling() {
    _pollTimer?.cancel();
    _pollTimer = null;
    state = state.copyWith(isPolling: false);
  }

  Future<void> _pollCycle() async {
    if (!state.isConnected || _ecuProfile == null) return;

    try {
      final updatedData = Map<String, double>.from(state.liveData);

      // 1. Читаем ПЕРВЫЙ блок (Обороты, Температура, Дроссель)
      final block1 = _ecuProfile!.blocks[0];
      final cmd1 = _isUdsM86 ? block1.udsCommand : block1.kwpCommand;
      final res1 = await _btService.sendCommand(cmd1, timeoutMs: 1000);
      final prefix1 = _isUdsM86 ? '622001' : '6101';
      final bytes1 = ObdParser.extractBytes(res1, prefix1);

      if (bytes1 != null && bytes1.isNotEmpty) {
        for (final param in block1.parameters) {
          final val = param.decode(bytes1);
          if (val != null) updatedData[param.id] = val;
        }
      }

      // 2. Читаем ВТОРОЙ блок (Пропуски, Напряжение УДК)
      if (_ecuProfile!.blocks.length > 1) {
        final block2 = _ecuProfile!.blocks[1];
        final cmd2 = _isUdsM86 ? block2.udsCommand : block2.kwpCommand;
        final res2 = await _btService.sendCommand(cmd2, timeoutMs: 1000);
        final prefix2 = _isUdsM86 ? '622002' : '6102';
        final bytes2 = ObdParser.extractBytes(res2, prefix2);

        if (bytes2 != null && bytes2.isNotEmpty) {
          for (final param in block2.parameters) {
            final val = param.decode(bytes2);
            if (val != null) updatedData[param.id] = val;
          }
        }
      }

      // 2.5. Читаем УДК и ДДК (стандартные OBD-II PID)
      final resO2 = await _btService.sendCommand('0114', timeoutMs: 800);
      final bytesO2 = ObdParser.extractBytes(resO2, '4114');
      if (bytesO2 != null && bytesO2.isNotEmpty) {
        updatedData['udk_volt'] = bytesO2[0] / 200.0;
      }

      final resO2b = await _btService.sendCommand('0115', timeoutMs: 800);
      final bytesO2b = ObdParser.extractBytes(resO2b, '4115');
      if (bytesO2b != null && bytesO2b.isNotEmpty) {
        updatedData['ddk_volt'] = bytesO2b[0] / 200.0;
      }

      // 3. Читаем СКОРОСТЬ для БК (стандартный OBD2 PID 010D)
      final resSpeed = await _btService.sendCommand('010D', timeoutMs: 800);
      final bytesSpeed = ObdParser.extractBytes(resSpeed, '410D');
      if (bytesSpeed != null && bytesSpeed.isNotEmpty) {
        updatedData['speed'] = bytesSpeed[0].toDouble(); // км/ч
      }

      // 4. Кормим данные в БК
      tripCalculator.update(
        rpm: updatedData['rpm']?.toInt() ?? 0,
        speedKmh: updatedData['speed'] ?? 0.0,
        airMassKgH: (updatedData['map'] ?? 35.0) * (updatedData['rpm'] ?? 840.0) * 0.00025,
        tpsPct: updatedData['tps'] ?? 0.0,
        isDfcoActive: false,
        stftPct: updatedData['stft'] ?? 0.0,
        ltftPct: updatedData['ltft'] ?? 0.0,
      );

      // РАСЧЕТ МАТРИЦЫ КОРРЕКЦИЙ
      final rpm = updatedData['rpm'] ?? 0;
      final tps = updatedData['tps'] ?? 0;
      final stft = updatedData['stft'];
      final ltft = updatedData['ltft'];

      if (stft != null && ltft != null && rpm > 0 && tps < 70) {
        if (rpm >= 740 && rpm <= 960 && tps <= 4.5) _bins['idle']!.addSample(stft, ltft);
        else if (rpm > 1600 && tps <= 12.0) _bins['low']!.addSample(stft, ltft);
        else if (rpm >= 1300 && rpm <= 3400 && tps > 4.5 && tps <= 35.0) _bins['part']!.addSample(stft, ltft);
        else if (tps > 35.0) _bins['high']!.addSample(stft, ltft);
        
        final medians = {
          'idle': _bins['idle']!.median,
          'low': _bins['low']!.median,
          'part': _bins['part']!.median,
          'high': _bins['high']!.median,
        };
        state = state.copyWith(trimMedians: medians);
      }

      // ЛОГИРОВАНИЕ АНОМАЛИЙ
      if (updatedData['ect'] != null && updatedData['ect']! > 103) {
        _logNuance('overheat', 'Перегрев ОЖ', 'Температура ${updatedData['ect']!.toInt()}°C > 103°C', NuanceSeverity.critical);
      }

      // Логирование коррекций
      if (updatedData['stft'] != null && updatedData['stft']!.abs() > 15) {
        _logNuance('trim_high', 'Коррекция STFT > 15%', 'STFT: ${updatedData['stft']!.toStringAsFixed(1)}%', NuanceSeverity.warning);
      }
      
      // Логирование просадки АКБ
      if (updatedData['volt'] != null && updatedData['volt']! < 10.5 && updatedData['volt']! > 5.0) {
        _logNuance('volt_low', 'Просадка АКБ', 'Напряжение: ${updatedData['volt']!.toStringAsFixed(2)} В', NuanceSeverity.critical);
      }

      // Обновляем график УДК
      final updatedTrace1 = List<double>.from(state.trace1);
      if (updatedData.containsKey('udk_volt')) {
        updatedTrace1.add(updatedData['udk_volt']!);
        if (updatedTrace1.length > 100) updatedTrace1.removeAt(0);
      }

      // Обновляем график ДДК
      final updatedTrace2 = List<double>.from(state.trace2);
      if (updatedData.containsKey('ddk_volt')) {
        updatedTrace2.add(updatedData['ddk_volt']!);
        if (updatedTrace2.length > 100) updatedTrace2.removeAt(0);
      }

      state = state.copyWith(liveData: updatedData, trace1: updatedTrace1, trace2: updatedTrace2);

    } catch (e) {
      // Ошибка опроса
    }
  }

  Future<List<String>> readDtcs() async {
    List<String> foundCodes = [];
    if (!state.isConnected) return foundCodes;

    try {
      // Пробуем стандартный OBD2 запрос (03), если не выйдет — шлем UDS (190208)
      String response = await _btService.sendCommand('03', timeoutMs: 2000);
      List<int>? bytes = ObdParser.extractBytes(response, '43');

      if (bytes == null) {
        // Если OBD2 не ответил, пробуем UDS
        response = await _btService.sendCommand('190208', timeoutMs: 2000);
        bytes = ObdParser.extractBytes(response, '590208');
      }

      if (bytes != null && bytes.isNotEmpty) {
        // Первый байт — количество ошибок. Дальше идут пары байтов (сами коды).
        int count = bytes[0];
        for (int i = 0; i < count; i++) {
          if (bytes.length >= 2 + (i * 2)) {
            int b1 = bytes[1 + (i * 2)];
            int b2 = bytes[2 + (i * 2)];
            if (b1 == 0 && b2 == 0) continue; // Пустые байты
            
            // Расшифровка DTC (P, C, B, U)
            const typeChars = ['P', 'C', 'B', 'U'];
            final type = typeChars[(b1 >> 6) & 0x03];
            final code = '$type${((b1 >> 4) & 0x03)}${(b1 & 0x0F).toRadixString(16).toUpperCase()}${((b2 >> 4) & 0x0F).toRadixString(16).toUpperCase()}${(b2 & 0x0F).toRadixString(16).toUpperCase()}';
            foundCodes.add(code);
          }
        }
      }
    } catch (e) {
      // Ошибка чтения
    }
    return foundCodes;
  }

    final rpm = state.liveData['rpm']?.toInt() ?? 0;

    int lastEct = 0;

    while (sw.elapsedMilliseconds < 30000) {
      if (!state.isConnected) return '⛔ СРЫВ: Потеря связи с ЭБУ!';
      
      final rVolt = await _btService.sendCommand('ATRV', timeoutMs: 500);
      final volt = double.tryParse(RegExp(r'(\d+\.\d+)').firstMatch(rVolt)?.group(1) ?? '12.0') ?? 12.0;

      final tpsBytes = ObdParser.extractBytes(rTps, '4111');
      if (tpsBytes != null && tpsBytes[0] * 100.0 / 255.0 > 9.5) return '⛔ СРЫВ: Педаль нажата!';

      if (sw.elapsedMilliseconds - lastEct > 5000) {
        final rEct = await _btService.sendCommand('0105', timeoutMs: 500);
        final ectBytes = ObdParser.extractBytes(rEct, '4105');
        if (ectBytes != null && (ectBytes[0] - 40) > 100) return '⛔ СРЫВ: ОЖ > 100°C!';
        lastEct = sw.elapsedMilliseconds;
      }

      onUpdate('Шаг 1/2: Осталось ${30 - (sw.elapsedMilliseconds ~/ 1000)} сек');
      await Future.delayed(const Duration(milliseconds: 250));
    }

    for (int i = 10; i > 0; i--) {

import 'dart:async';
import 'package:flutter_bluetooth_classic_serial/flutter_bluetooth_classic.dart';

class BluetoothService {
  final FlutterBluetoothClassic _bt = FlutterBluetoothClassic();
  final StringBuffer _rxBuffer = StringBuffer();
  
  Completer<void>? _cmdLock;
  bool _isConnected = false;
  StreamSubscription? _dataSubscription;

  bool get isConnected => _isConnected;

  Future<List<Map<String, String>>> getPairedDevices() async {
    try {
      final devices = await _bt.getPairedDevices();
      return devices.map((d) => {'name': d.name, 'address': d.address}).toList();
    } catch (e) {
      return [];
    }
  }

  Future<bool> connect(String address) async {
    try {
      await _bt.connect(address);
      _isConnected = true;
      _dataSubscription = _bt.onDataReceived.listen((data) {
        _rxBuffer.write(String.fromCharCodes(data.data));
      });
      return true;
    } catch (e) {
      _isConnected = false;
      return false;
    }
  }

  Future<void> disconnect() async {
    try {
      await _dataSubscription?.cancel();
      await _bt.disconnect();
    } finally {
      _isConnected = false;
    }
  }

  Future<String> sendCommand(String cmd, {int timeoutMs = 1200, int postDelayMs = 50}) async {
    if (!_isConnected) return 'ERR:NOT_CONNECTED';
    while (_cmdLock != null) {
      await _cmdLock!.future;
    }
    final lock = Completer<void>();
    _cmdLock = lock;

    try {
      _rxBuffer.clear();
      await _bt.sendString('$cmd\r');
      final sw = Stopwatch()..start();
      while (sw.elapsedMilliseconds < timeoutMs) {
        await Future.delayed(const Duration(milliseconds: 20));
        final current = _rxBuffer.toString();
        if (current.contains('>')) {
          if (postDelayMs > 0) await Future.delayed(Duration(milliseconds: postDelayMs));
          return current.replaceAll('>', '').replaceAll('\r', '\n').trim();
        }
      }
      return _rxBuffer.toString().trim();
    } catch (e) {
      return 'ERR:$e';
    } finally {
      _cmdLock = null;
      lock.complete();
    }
  }
}

// ISO 15765-2 парсер. Не блокирует кадры с байтом 7F внутри данных.
// Отсекает только реальный Negative Response (7F + код сервиса в начале).
class ObdParser {
  static List<int>? extractBytes(String resp, String expectedPrefix) {
    final upper = resp.toUpperCase();
    if (upper.contains('NO DATA') ||
        upper.contains('ERROR') ||
        upper.contains('UNABLE') ||
        upper.contains('STOPPED') ||
        upper.contains('?')) {
      return null;
    }

    int? expectedTotalLen;
    final lines = upper.split(RegExp(r'[\r\n]+'));
    final cleanHexBuf = StringBuffer();

    for (int lineIdx = 0; lineIdx < lines.length; lineIdx++) {
      var line = lines[lineIdx].trim();
      if (line.isEmpty || line.startsWith('SEARCHING')) continue;

      // 12-битный заголовок длины ISO-TP («01A», «024») идёт отдельной первой строкой
      if (lineIdx == 0 &&
          line.length == 3 &&
          RegExp(r'^[0-9A-F]{3}$').hasMatch(line)) {
        expectedTotalLen = int.tryParse(line, radix: 16);
        continue;
      }

      // Отрезаем префикс номера кадра ISO-TP ("0:", "1:", "2:", ...)
      final colonIdx = line.indexOf(':');
      if (colonIdx != -1 && colonIdx <= 2) {
        line = line.substring(colonIdx + 1).trim();
      }

      // Отрезаем CAN-ID заголовки вида 7E8
      if (line.startsWith('7E8')) {
        line = line.substring(3).trim();
      }

      final hexOnly = line.replaceAll(RegExp(r'[^0-9A-F]'), '');
      cleanHexBuf.write(hexOnly);
    }

    final clean = cleanHexBuf.toString();
    if (clean.isEmpty) return null;

    // Проверка Negative Response только в начале ответа
    final pref = expectedPrefix.toUpperCase();
    if (pref.length >= 2) {
      final respServiceByte = int.tryParse(pref.substring(0, 2), radix: 16);
      if (respServiceByte != null && respServiceByte >= 0x40) {
        final reqServiceHex = (respServiceByte - 0x40)
            .toRadixString(16)
            .padLeft(2, '0')
            .toUpperCase();
        if (clean.startsWith('7F$reqServiceHex') ||
            clean.startsWith('037F$reqServiceHex')) {
          return null;
        }
      }
    }

    final idx = clean.indexOf(pref);
    if (idx == -1) return null;

    final hexData = clean.substring(idx + pref.length);
    final bytes = <int>[];
    for (int i = 0; i + 1 < hexData.length; i += 2) {
      final b = int.tryParse(hexData.substring(i, i + 2), radix: 16);
      if (b != null) bytes.add(b);
    }

    final payloadMaxLen = expectedTotalLen != null
        ? (expectedTotalLen - (pref.length ~/ 2))
        : null;
    if (payloadMaxLen != null &&
        payloadMaxLen > 0 &&
        bytes.length > payloadMaxLen) {
      return bytes.sublist(0, payloadMaxLen);
    }
    return bytes;
  }

  static int? tryParseMode06Counter(String resp, String expectedPrefix) {
    final b = extractBytes(resp, expectedPrefix);
    if (b == null || b.length < 4) return null;
    if (b.length >= 5) return (b[3] * 256) + b[4];
    return (b[2] * 256) + b[3];
  }
}

import 'dart:convert';
import 'dart:io';

class OtaService {
  // Константы твоего репозитория
  static const String currentVersion = '2.2.0';
  static const String githubRepo = 'MrValenOK/lada_vesta_scanner';

  static Future<Map<String, dynamic>?> checkForUpdates() async {
    final client = HttpClient()
      ..connectionTimeout = const Duration(seconds: 5); // Уменьшили с 8

    try {
      final req = await client.getUrl(
        Uri.parse('https://api.github.com/repos/$githubRepo/releases/latest'),
      );
      req.headers.set('User-Agent', 'VestaScanPro-App');
      req.headers.set('Accept', 'application/vnd.github+json');
      
      final resp = await req.close();

      if (resp.statusCode == 200) {
        final bodyStr = await resp.transform(utf8.decoder).join();
        final data = jsonDecode(bodyStr) as Map<String, dynamic>;

        final tagName = (data['tag_name'] ?? currentVersion).toString();
        final notes = (data['body'] ?? '').toString().trim();
        
        String apkUrl = 'https://github.com/$githubRepo/releases/latest/download/app-release.apk';
        if (data['assets'] is List && (data['assets'] as List).isNotEmpty) {
          for (final asset in (data['assets'] as List)) {
            final name = (asset['name'] ?? '').toString().toLowerCase();
            if (name.endsWith('.apk') && asset['browser_download_url'] != null) {
              apkUrl = asset['browser_download_url'].toString();
              break;
            }
          }
        }

        final isNewer = _isVersionHigher(tagName, currentVersion);

        return {
          'hasUpdate': isNewer,
          'latestVersion': tagName,
          'releaseNotes': notes.isNotEmpty ? notes : 'Обновление алгоритмов диагностики.',
          'apkUrl': apkUrl,
        };
      }
    } catch (e) {
      // Не молчим — возвращаем специальный маркер, что API недоступен,
      // но даем возможность открыть страницу релизов напрямую.
      return {
        'hasUpdate': false,
        'error': 'API_UNREACHABLE',
        'fallbackUrl': 'https://github.com/$githubRepo/releases/latest',
      };
    } finally {
      client.close(force: true);
    }
    return {
      'hasUpdate': false,
      'error': 'API_UNREACHABLE',
      'fallbackUrl': 'https://github.com/$githubRepo/releases/latest',
    };
  }

  // Сравнение версий (например, v2.2.0 > v2.1.0)
  static bool _isVersionHigher(String remote, String local) {
    String clean(String v) => v.split('+').first.split('-').first.replaceAll(RegExp(r'[^0-9.]'), '');
    
    final rParts = clean(remote).split('.').map((e) => int.tryParse(e) ?? 0).toList();
    final lParts = clean(local).split('.').map((e) => int.tryParse(e) ?? 0).toList();
    
    final maxLen = rParts.length > lParts.length ? rParts.length : lParts.length;
    for (int i = 0; i < maxLen; i++) {
      final r = i < rParts.length ? rParts[i] : 0;
      final l = i < lParts.length ? lParts[i] : 0;
      if (r > l) return true;
      if (r < l) return false;
    }
    return false;
  }
}

class TripCalculator {
  double totalDistanceKm = 0.0;
  double totalFuelLiters = 0.0;
  double maxSpeedKmh = 0.0;

  // Текущие показатели для вывода на дашборд в реальном времени
  double instantFuelLpH = 0.0;
  double instantFuelL100Km = 0.0;

  DateTime? _lastTickTime;

  // Константы для бензина АИ-92 / АИ-95
  final double stoichiometricRatio = 14.7;
  final double fuelDensityKgL = 0.75; // ~0.75 кг/л при 15-20°C

  void update({
    required int rpm,
    required double speedKmh,
    required double airMassKgH,
    required double tpsPct,
    required bool isDfcoActive,
    double stftPct = 0.0, // Краткосрочная коррекция М86 (%)
    double ltftPct = 0.0, // Долгосрочная коррекция М86 (%)
  }) {
    final now = DateTime.now();
    if (_lastTickTime == null) {
      _lastTickTime = now;
      return;
    }

    final deltaTimeSec = now.difference(_lastTickTime!).inMilliseconds / 1000.0;
    final deltaHours = deltaTimeSec / 3600.0;
    _lastTickTime = now;

    // Защита от зависания Bluetooth или паузы опроса
    if (deltaTimeSec <= 0.0 || deltaTimeSec > 5.0) return;

    // 1. Интеграл скорости (пробег)
    if (speedKmh > 0) {
      totalDistanceKm += speedKmh * deltaHours;
      if (speedKmh > maxSpeedKmh) maxSpeedKmh = speedKmh;
    }

    // 2. Расход топлива с защитой ПХХ (отсечка топливоподачи)
    double currentFuelKgH = 0.0;

    // Если ЭБУ остановил двигатель (rpm == 0) или активен режим ПХХ — подача 0
    final bool isCutoff = (rpm < 300) || isDfcoActive || (tpsPct < 2.0 && rpm > 1500);

    if (!isCutoff && airMassKgH > 0) {
      // Коэффициент коррекции смеси по УДК (например, STFT -3.9% и LTFT +0.8%)
      final double trimFactor = (1.0 + (stftPct / 100.0)) * (1.0 + (ltftPct / 100.0));
      currentFuelKgH = (airMassKgH / stoichiometricRatio) * trimFactor;
    }

    // Перевод из кг/ч в л/ч
    instantFuelLpH = currentFuelKgH / fuelDensityKgL;
    totalFuelLiters += instantFuelLpH * deltaHours;

    // Мгновенный путевой расход (считаем только при движении > 5 км/ч)
    if (speedKmh >= 5.0) {
      instantFuelL100Km = (instantFuelLpH / speedKmh) * 100.0;
    } else {
      instantFuelL100Km = 0.0; // На месте показываем instantFuelLpH (л/ч)
    }
  }

  // Средний расход на 100 км за поездку
  double get averageConsumptionL100Km {
    if (totalDistanceKm < 0.1) return 0.0;
    return (totalFuelLiters / totalDistanceKm) * 100.0;
  }

  void reset() {
    totalDistanceKm = 0.0;
    totalFuelLiters = 0.0;
    maxSpeedKmh = 0.0;
    instantFuelLpH = 0.0;
    instantFuelL100Km = 0.0;
    _lastTickTime = null;
  }
}
