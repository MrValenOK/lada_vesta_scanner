import 'dart:math' as math;

class EcuParameter {
  final String id;
  final String name;
  final int byteOffset; // Смещение байта в ответе ЭБУ
  final int width;      // Количество байт (обычно 1 или 2)
  final double multiplier;
  final double divisor;
  final double offset;
  final int decimals;   // Количество знаков после запятой
  final bool signedComplement; // Использовать ли дополнительный код (Two's complement) для отрицательных чисел
  final int bitIndex;   // Если -1, то это число. Если >= 0, то это битовый флаг (0 или 1)
  final String unit;    // Единица измерения (например, "°C" или "%")

  EcuParameter({
    required this.id,
    required this.name,
    required this.byteOffset,
    required this.width,
    this.multiplier = 1.0,
    this.divisor = 1.0,
    this.offset = 0.0,
    this.decimals = 0,
    this.signedComplement = false,
    this.bitIndex = -1,
    this.unit = '',
  });

  factory EcuParameter.fromJson(Map<String, dynamic> json) {
    return EcuParameter(
      id: json['id'] as String,
      name: json['name'] as String,
      byteOffset: json['byteOffset'] as int,
      width: json['width'] as int? ?? 1,
      multiplier: (json['multiplier'] as num?)?.toDouble() ?? 1.0,
      divisor: (json['divisor'] as num?)?.toDouble() ?? 1.0,
      offset: (json['offset'] as num?)?.toDouble() ?? 0.0,
      decimals: json['decimals'] as int? ?? 0,
      signedComplement: json['signedComplement'] as bool? ?? false,
      bitIndex: json['bitIndex'] as int? ?? -1,
      unit: json['unit'] as String? ?? '',
    );
  }

  /// Метод декодирования сырых байт в готовое значение
  double? decode(List<int> payload) {
    // Проверка выхода за границы массива
    if (byteOffset < 0 || byteOffset + width > payload.length) return null;

    int raw = 0;
    // Собираем число из байтов (Big Endian)
    for (int i = 0; i < width; i++) {
      raw = (raw << 8) | payload[byteOffset + i];
    }

    // Если это битовый флаг
    if (bitIndex >= 0) {
      return ((raw >> bitIndex) & 1).toDouble();
    }

    // Обработка отрицательных чисел (Дополнительный код / Two's complement)
    if (signedComplement) {
      final signBit = 1 << (width * 8 - 1);
      if ((raw & signBit) != 0) {
        raw -= (1 << (width * 8));
      }
    }

    // Применяем формулу: (raw * multiplier / divisor) + offset
    double v = raw * multiplier / divisor + offset;
    
    // Округляем до нужного количества знаков
    final factor = math.pow(10, decimals);
    v = (v * factor).roundToDouble() / factor;
    
    // Избегаем "-0.0"
    return v == 0 ? 0.0 : v;
  }
}

import 'ecu_parameter.dart';

class EcuRequestBlock {
  final String name;       // Название пакета (например, "Основные параметры")
  final String kwpCommand; // Команда для старых блоков (например, "2101")
  final String udsCommand; // Команда для новых блоков UDS (например, "220001" или "222001")
  final List<EcuParameter> parameters;

  EcuRequestBlock({
    required this.name,
    required this.kwpCommand,
    required this.udsCommand,
    required this.parameters,
  });

  factory EcuRequestBlock.fromJson(Map<String, dynamic> json) {
    return EcuRequestBlock(
      name: json['name'] as String,
      kwpCommand: json['kwpCommand'] as String,
      udsCommand: json['udsCommand'] as String,
      parameters: (json['parameters'] as List)
          .map((p) => EcuParameter.fromJson(p as Map<String, dynamic>))
          .toList(),
    );
  }
}

class EcuProfile {
  final String id;
  final String name;
  final List<EcuRequestBlock> blocks;

  EcuProfile({
    required this.id,
    required this.name,
    required this.blocks,
  });

  factory EcuProfile.fromJson(Map<String, dynamic> json) {
    return EcuProfile(
      id: json['id'] as String,
      name: json['name'] as String,
      blocks: (json['blocks'] as List)
          .map((b) => EcuRequestBlock.fromJson(b as Map<String, dynamic>))
          .toList(),
    );
  }
}

import 'package:flutter/material.dart';
import 'dart:collection';

class Oscilloscope extends StatelessWidget {
  final Queue<double> trace1;
  final Queue<double>? trace2;
  final double minValue;
  final double maxValue;
  final Color color1;
  final Color? color2;

  const Oscilloscope({
    super.key,
    required this.trace1,
    this.trace2,
    required this.minValue,
    required this.maxValue,
    required this.color1,
    this.color2,
  });

  @override
  Widget build(BuildContext context) {
    return Container(
      height: 180,
      width: double.infinity,
      decoration: BoxDecoration(
        color: const Color(0xFF10151D), // darkScopeBg из нашей темы
        borderRadius: BorderRadius.circular(16),
        border: Border.all(color: Theme.of(context).dividerColor),
      ),
      child: ClipRRect(
        borderRadius: BorderRadius.circular(16),
        // RepaintBoundary спасает батарею, изолируя перерисовку
        child: RepaintBoundary(
          child: CustomPaint(
            painter: _ScopePainter(
              trace1: trace1.toList(),
              trace2: trace2?.toList(),
              minVal: minValue,
              maxVal: maxValue,
              color1: color1,
              color2: color2 ?? Colors.transparent,
              gridColor: const Color(0xFF243040),
            ),
          ),
        ),
      ),
    );
  }
}

class _ScopePainter extends CustomPainter {
  final List<double> trace1;
  final List<double>? trace2;
  final double minVal;
  final double maxVal;
  final Color color1;
  final Color color2;
  final Color gridColor;

  _ScopePainter({
    required this.trace1,
    this.trace2,
    required this.minVal,
    required this.maxVal,
    required this.color1,
    required this.color2,
    required this.gridColor,
  });

  @override
  void paint(Canvas canvas, Size size) {
    // 1. Отрисовка фоновой сетки (Grid)
    final gridPaint = Paint()
      ..color = gridColor
      ..strokeWidth = 1.0;
    
    // Горизонтальные линии
    for (int i = 1; i < 4; i++) {
      final y = size.height * (i / 4);
      canvas.drawLine(Offset(0, y), Offset(size.width, y), gridPaint);
    }
    // Вертикальные линии
    for (int i = 1; i < 6; i++) {
      final x = size.width * (i / 6);
      canvas.drawLine(Offset(x, 0), Offset(x, size.height), gridPaint);
    }

    // 2. Логика отрисовки самой кривой
    void drawTrace(List<double> trace, Color color) {
      if (trace.isEmpty) return;
      final paint = Paint()
        ..color = color
        ..strokeWidth = 2.0
        ..style = PaintingStyle.stroke
        ..strokeJoin = StrokeJoin.round;

      final path = Path();
      // Выделяем ширину шага, предполагая, что на экране хранится до 100 точек
      final stepX = size.width / 99; 
      final range = maxVal - minVal;
      if (range == 0) return;

      for (int i = 0; i < trace.length; i++) {
        final x = i * stepX;
        // Нормализация (от 0.0 до 1.0) и инверсия (ось Y во Flutter идет сверху вниз)
        double normalizedY = (trace[i] - minVal) / range;
        normalizedY = normalizedY.clamp(0.0, 1.0);
        final y = size.height - (normalizedY * size.height);

        if (i == 0) {
          path.moveTo(x, y);
        } else {
          path.lineTo(x, y);
        }
      }
      canvas.drawPath(path, paint);
    }

    // Рисуем графики
    drawTrace(trace1, color1);
    if (trace2 != null) {
      drawTrace(trace2!, color2);
    }
  }

  @override
  bool shouldRepaint(covariant _ScopePainter oldDelegate) => true; 
}

import 'dart:async';
import 'dart:math' as math;
import 'package:flutter_riverpod/flutter_riverpod.dart';
import '../services/bluetooth_service.dart';

class ObdState {
  final bool isConnected;
  final bool isPolling;
  final Map<String, double> liveData;
  final List<double> trace1;

  ObdState({
    this.isConnected = false,
    this.isPolling = false,
    this.liveData = const {},
    this.trace1 = const [],
  });

  ObdState copyWith({
    bool? isConnected,
    bool? isPolling,
    Map<String, double>? liveData,
    List<double>? trace1,
  }) {
    return ObdState(
      isConnected: isConnected ?? this.isConnected,
      isPolling: isPolling ?? this.isPolling,
      liveData: liveData ?? this.liveData,
      trace1: trace1 ?? this.trace1,
    );
  }
}

final bluetoothServiceProvider = Provider((ref) => BluetoothService());

final obdProvider = StateNotifierProvider<ObdNotifier, ObdState>((ref) {
  return ObdNotifier(ref.read(bluetoothServiceProvider));
});

class ObdNotifier extends StateNotifier<ObdState> {
  final BluetoothService _btService;
  Timer? _pollTimer;
  bool _isPollCycleBusy = false;
  double _demoSineWave = 0.0;

  ObdNotifier(this._btService) : super(ObdState());

  Future<bool> connect(String macAddress) async {
    state = state.copyWith(isConnected: true);
    return true;
  }

  Future<void> disconnect() async {
    stopPolling();
    await _btService.disconnect();
    state = state.copyWith(isConnected: false, liveData: {}, trace1: []);
  }

  void startPolling() {
    if (!state.isConnected || state.isPolling) return;
    state = state.copyWith(isPolling: true);

    _pollTimer = Timer.periodic(const Duration(milliseconds: 100), (timer) async {
      await _pollCycle();
    });
  }

  void stopPolling() {
    _pollTimer?.cancel();
    _pollTimer = null;
    state = state.copyWith(isPolling: false);
  }

  Future<void> _pollCycle() async {
    if (_isPollCycleBusy) return;
    _isPollCycleBusy = true;

    try {
      _demoSineWave += 0.2;
      final double simulatedValue = (math.sin(_demoSineWave) * 400) + 800;

      final updatedData = Map<String, double>.from(state.liveData);
      updatedData['rpm'] = simulatedValue;
      updatedData['ect'] = 88.0;

      final updatedTrace = List<double>.from(state.trace1);
      updatedTrace.add(simulatedValue);
      if (updatedTrace.length > 100) {
        updatedTrace.removeAt(0);
      }

      state = state.copyWith(liveData: updatedData, trace1: updatedTrace);
    } finally {
      _isPollCycleBusy = false;
    }
  }
}

class AiPromptBuilder {
  static String generateSnapshot({
    required Map<String, double> liveData,
    required double tripDistance,
    required double fuelConsumed,
    String userNotes = "",
  }) {
    final buffer = StringBuffer();
    
    buffer.writeln('Действуй как эксперт-автодиагност. Проанализируй лог телеметрии автомобиля Lada Vesta (ЭБУ Итэлма М86 UDS) и найди возможные аномалии.');
    if (userNotes.isNotEmpty) {
      buffer.writeln('Жалоба водителя: "$userNotes"');
    }
    
    buffer.writeln('\n--- ДАТЧИКИ ДВИГАТЕЛЯ (ТЕКУЩИЕ) ---');
    buffer.writeln('Обороты: ${liveData["rpm"]?.toInt() ?? 0} об/мин');
    buffer.writeln('Температура ОЖ: ${liveData["ect"]?.toInt() ?? 0} °C');
    
    // Безопасное извлечение коррекций, если они есть
    if (liveData.containsKey('stft')) {
      buffer.writeln('Краткосрочная коррекция (STFT): ${liveData["stft"]}%');
    }
    if (liveData.containsKey('ltft')) {
      buffer.writeln('Долгосрочная коррекция (LTFT): ${liveData["ltft"]}%');
    }

    buffer.writeln('\n--- ДАННЫЕ БОРТОВОГО КОМПЬЮТЕРА ---');
    buffer.writeln('Длительность текущей поездки: ${tripDistance.toStringAsFixed(1)} км');
    buffer.writeln('Потрачено топлива: ${fuelConsumed.toStringAsFixed(2)} л');

    buffer.writeln('\nТвоя задача:');
    buffer.writeln('1. Оцени адекватность показаний датчиков для текущего режима.');
    buffer.writeln('2. Если есть отклонения, назови 3 наиболее вероятные аппаратные причины.');
    buffer.writeln('3. Какие параметры мне нужно проверить дополнительно, чтобы сузить поиск?');

    return buffer.toString();
  }
}

class BluetoothService {
  final FlutterBluetoothClassic _bt = FlutterBluetoothClassic();
  final StringBuffer _rxBuffer = StringBuffer();
  
  Completer<void>? _cmdLock;
  bool _isConnected = false;
  StreamSubscription? _dataSubscription;

  bool get isConnected => _isConnected;

  /// Инициализация и подключение к ELM327
  Future<bool> connect(String address) async {
    try {
      await _bt.connect(address);
      _isConnected = true;
      
      // Подписка на входящий поток байтов
      _dataSubscription = _bt.onDataReceived.listen((data) {
        // Конвертируем байты в строку и добавляем в буфер
        _rxBuffer.write(String.fromCharCodes(data.data));
      });
      return true;
    } catch (e) {
      _isConnected = false;
      return false;
    }
  }

  /// Отключение
  Future<void> disconnect() async {
    try {
      await _dataSubscription?.cancel();
      await _bt.disconnect();
    } finally {
      _isConnected = false;
    }
  }

  /// Главный метод отправки команд с защитой от одновременных запросов
  Future<String> sendCommand(
    String cmd, {
    int timeoutMs = 1200,
    int postDelayMs = 50,
  }) async {
    if (!_isConnected) return 'ERR:NOT_CONNECTED';

    // Очередь команд: ждем, если шина занята
    while (_cmdLock != null) {
      await _cmdLock!.future;
    }
    
    final lock = Completer<void>();
    _cmdLock = lock;

    try {
      _rxBuffer.clear();
      // Отправляем команду с символом возврата каретки (требование ELM327)
      await _bt.sendString('$cmd\r');

      final sw = Stopwatch()..start();
      while (sw.elapsedMilliseconds < timeoutMs) {
        await Future.delayed(const Duration(milliseconds: 20));
        final current = _rxBuffer.toString();
        
        // Как только прилетел символ '>', адаптер закончил передачу
        if (current.contains('>')) {
          if (postDelayMs > 0) {
            await Future.delayed(Duration(milliseconds: postDelayMs));
          }
          // Очищаем ответ от символа '>' и лишних пробелов
          return current.replaceAll('>', '').replaceAll('\r', '\n').trim();
        }
      }
      
      // Если уперлись в таймаут, отдаем то, что успели получить
      return _rxBuffer.toString().trim();
    } catch (e) {
      return 'ERR:$e';
    } finally {
      _cmdLock = null;
      lock.complete();
    }
  }
}

class ObdParser {
  /// Проверяет, является ли ответ отрицательным (NRC - Negative Response Code)
  static bool isNegativeResponse(String resp) {
    final upper = resp.toUpperCase().trim();
    if (upper.isEmpty || 
        upper.contains('NO DATA') || 
        upper.contains('ERROR') || 
        upper.contains('UNABLE') || 
        upper.contains('?')) {
      return true;
    }
    final clean = upper.replaceAll(RegExp(r'[^0-9A-F]'), '');
    if (clean.isEmpty) return true;
    // 7F - маркер ошибки протокола (например, 7F 21 11)
    return RegExp(r'^(03)?7F[0-9A-F]{4}').hasMatch(clean);
  }

  /// Извлекает чистые байты из ответа ELM327, игнорируя эхо, заголовки и мусор.
  /// expectedPrefix - ожидаемый ответ (например, "620001" на запрос "220001")
  static List<int>? extractBytes(String resp, String expectedPrefix) {
    if (isNegativeResponse(resp)) return null;

    final lines = resp.toUpperCase().split(RegExp(r'[\r\n]+'));
    final cleanHexBuf = StringBuffer();

    for (var line in lines) {
      line = line.trim();
      if (line.isEmpty || line.startsWith('SEARCHING')) continue;
      
      // Очищаем от CAN-заголовков (например, 7E8 для двигателя)
      if (line.startsWith('7E8')) {
        line = line.substring(3).trim();
      }
      
      // Очищаем от нумерации строк многокадровых сообщений (0:, 1: и т.д.)
      final colonIdx = line.indexOf(':');
      if (colonIdx != -1 && colonIdx <= 2) {
        line = line.substring(colonIdx + 1).trim();
      }

      // Оставляем только HEX-символы
      final hexOnly = line.replaceAll(RegExp(r'[^0-9A-F]'), '');
      cleanHexBuf.write(hexOnly);
    }

    final clean = cleanHexBuf.toString();
    final pref = expectedPrefix.toUpperCase();
    
    // Ищем начало полезной нагрузки
    final idx = clean.indexOf(pref);
    if (idx == -1) return null; 

    // Достаем всё, что идет после префикса
    final hexData = clean.substring(idx + pref.length);
    final bytes = <int>[];
    
    // Переводим текст в массив байтов
    for (int i = 0; i + 1 < hexData.length; i += 2) {
      final b = int.tryParse(hexData.substring(i, i + 2), radix: 16);
      if (b != null) bytes.add(b);
    }
    
    return bytes;
  }
}

class TripCalculator {
  double totalDistanceKm = 0.0;
  double totalFuelLiters = 0.0;
  double maxSpeedKmh = 0.0;
  
  DateTime? _lastTickTime;
  
  // Константы
  final double stoichiometricRatio = 14.7;
  final double fuelDensityKgL = 0.75;

  void update({
    required int rpm,
    required double speedKmh,
    required double airMassKgH,
    required double tpsPct,
    required bool isDfcoActive,
  }) {
    final now = DateTime.now();
    
    if (_lastTickTime == null) {
      _lastTickTime = now;
      return;
    }
    
    final deltaTimeSec = now.difference(_lastTickTime!).inMilliseconds / 1000.0;
    final deltaHours = deltaTimeSec / 3600.0;
    _lastTickTime = now;

    // Защита от слишком долгих пауз (например, потеря связи Bluetooth)
    if (deltaTimeSec > 5.0) return;

    // 1. Интеграл скорости (пробег)
    if (speedKmh > 0) {
      totalDistanceKm += speedKmh * deltaHours;
      if (speedKmh > maxSpeedKmh) maxSpeedKmh = speedKmh;
    }

    // 2. Расчет расхода топлива с защитой ПХХ (отсечка)
    double currentFuelKgH = 0.0;
    final bool isCutoff = isDfcoActive || (tpsPct < 2.0 && rpm > 1500);

    if (!isCutoff) {
      currentFuelKgH = airMassKgH / stoichiometricRatio;
    }

    final currentFuelLpH = currentFuelKgH / fuelDensityKgL;
    totalFuelLiters += currentFuelLpH * deltaHours;
  }

  double get averageConsumptionL100Km {
    if (totalDistanceKm < 0.1) return 0.0;
    return (totalFuelLiters / totalDistanceKm) * 100;
  }
  
  void reset() {
    totalDistanceKm = 0.0;
    totalFuelLiters = 0.0;
    maxSpeedKmh = 0.0;
    _lastTickTime = null;
  }
}
